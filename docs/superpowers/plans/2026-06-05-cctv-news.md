> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add CCTV News (央视新闻) as a 4th intel category with RSS collection via RSSHub, AI-enhanced processing, statistical dashboards, event detection, and news-market correlation analysis.

**Architecture:** Extend the existing intel infrastructure (BaseNewsProvider → Collector → Repository → Scheduler) with a new `CCTVNewsProvider` and a dedicated `analysis/` submodule for stats, event detection, and market correlation. A new `intel_cctv_router` exposes dedicated endpoints. Frontend adds a 4th tab to the Intel page.

**Tech Stack:** Python 3.12 / FastAPI / SQLModel ORM / PostgreSQL (JSONB, ARRAY) / APScheduler / feedparser / React 19 + TypeScript / CSS BEM

```python
"""CCTV News provider — RSS feeds via RSSHub with channel metadata."""

from datetime import datetime, timezone

import feedparser
from loguru import logger

from src.domain.market.intel.models import NewsItem
from src.domain.market.intel.providers.base import BaseNewsProvider


class CCTVNewsProvider(BaseNewsProvider):
    """Collects CCTV news from RSSHub feeds with channel metadata."""

    name = "CCTV News"
    category = "cctv_news"
    source_type = "rss"

    def __init__(self, feeds: list[dict]):
        """
        Args:
            feeds: List of feed configs, each with keys:
                name, url, channel (domestic/world/finance/tech)
        """
        self.feeds = feeds

    def fetch_latest(self, limit: int = 50) -> list[NewsItem]:
        items: list[NewsItem] = []
        for feed_conf in self.feeds:
            url = feed_conf.get("url", "")
            channel = feed_conf.get("channel", "unknown")
            feed_name = feed_conf.get("name", url)
            if not url:
                continue
            try:
                feed = feedparser.parse(url)
                for entry in feed.entries[:limit]:
                    published = None
                    if hasattr(entry, "published_parsed") and entry.published_parsed:
                        published = datetime(
                            *entry.published_parsed[:6],
                            tzinfo=timezone.utc,
                        )
                    elif hasattr(entry, "updated_parsed") and entry.updated_parsed:
                        published = datetime(
                            *entry.updated_parsed[:6],
                            tzinfo=timezone.utc,
                        )

                    content = ""
                    if hasattr(entry, "summary"):
                        content = entry.summary
                    elif hasattr(entry, "content") and entry.content:
                        content = entry.content[0].get("value", "")

                    items.append(NewsItem(
                        title=entry.get("title", ""),
                        url=entry.get("link", ""),
                        source=feed_name,
                        source_type="rss",
                        category="cctv_news",
                        language="zh",
                        content=content[:2000],
                        published_at=published,
                        metadata={
                            "channel": channel,
                        },
                    ))
            except Exception as e:
                logger.warning(
                    f"[CCTVNewsProvider] Failed to fetch {url}: {e}")

        items.sort(
            key=lambda x: x.published_at
            or datetime.min.replace(tzinfo=timezone.utc),
            reverse=True,
        )
        return items[:limit]
```

    def get_cctv_stats(
        self, period: str = "7d", granularity: str = "day",
    ) -> dict:
        days_map = {"7d": 7, "30d": 30, "90d": 90}
        days = days_map.get(period, 7)
        with self._db.session_scope() as session:
            # Total count
            result = session.execute(text("""
                SELECT COUNT(*) as total_count
                FROM intel_news
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
            """), {"days": days})
            total = result.fetchone()._mapping["total_count"]

            # By channel
            result = session.execute(text("""
                SELECT metadata->>'channel' as channel, COUNT(*) as count
                FROM intel_news
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
                  AND metadata IS NOT NULL
                GROUP BY metadata->>'channel'
            """), {"days": days})
            by_channel = {row._mapping["channel"]: row._mapping["count"]
                          for row in result.fetchall()}

            # By sentiment
            result = session.execute(text("""
                SELECT
                    CASE
                        WHEN sentiment > 0.2 THEN 'positive'
                        WHEN sentiment < -0.2 THEN 'negative'
                        ELSE 'neutral'
                    END as bucket,
                    COUNT(*) as count
                FROM intel_news
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
                  AND sentiment IS NOT NULL
                GROUP BY bucket
            """), {"days": days})
            by_sentiment = {row._mapping["bucket"]: row._mapping["count"]
                            for row in result.fetchall()}

            # Sentiment trend by day
            result = session.execute(text("""
                SELECT
                    DATE(published_at) as date,
                    SUM(CASE WHEN sentiment > 0.2 THEN 1 ELSE 0 END) as positive,
                    SUM(CASE WHEN sentiment BETWEEN -0.2 AND 0.2 THEN 1 ELSE 0 END) as neutral,
                    SUM(CASE WHEN sentiment < -0.2 THEN 1 ELSE 0 END) as negative
                FROM intel_news
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
                  AND sentiment IS NOT NULL
                GROUP BY DATE(published_at)
                ORDER BY date
            """), {"days": days})
            sentiment_trend = [
                {
                    "date": str(row._mapping["date"]),
                    "positive": row._mapping["positive"],
                    "neutral": row._mapping["neutral"],
                    "negative": row._mapping["negative"],
                }
                for row in result.fetchall()
            ]

            # Top tags
            result = session.execute(text("""
                SELECT t as tag, COUNT(*) as count
                FROM intel_news, unnest(tags) t
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
                GROUP BY t
                ORDER BY count DESC
                LIMIT 15
            """), {"days": days})
            top_tags = [
                {"tag": row._mapping["tag"], "count": row._mapping["count"]}
                for row in result.fetchall()
            ]

            # Daily counts
            result = session.execute(text("""
                SELECT DATE(published_at) as date, COUNT(*) as count
                FROM intel_news
                WHERE category = 'cctv_news'
                  AND published_at >= NOW() - (:days || ' days')::interval
                GROUP BY DATE(published_at)
                ORDER BY date
            """), {"days": days})
            daily_counts = [
                {"date": str(row._mapping["date"]), "count": row._mapping["count"]}
                for row in result.fetchall()
            ]

        return {
            "period": period,
            "total_count": total,
            "by_channel": by_channel,
            "by_sentiment": by_sentiment,
            "sentiment_trend": sentiment_trend,
            "top_tags": top_tags,
            "daily_counts": daily_counts,
        }

    def upsert_event(self, event: dict) -> str:
        from sqlalchemy.dialects.postgresql import insert as pg_insert

                start_time=event.get("start_time"),
                last_seen=event.get("last_seen"),
                heat_score=event.get("heat_score", 0.0),
                trend=event.get("trend", "rising"),
            ).on_conflict_do_update(
                index_elements=["id"],
                set_={
                    "title": event["title"],
                    "keywords": event.get("keywords", []),
                    "news_ids": event.get("news_ids", []),
                    "last_seen": event.get("last_seen"),
                    "heat_score": event.get("heat_score", 0.0),

"""Hot topic event detection for CCTV news.

Algorithm:
1. Burst detection: Monitor 2-hour windows for abnormal volume
2. Topic clustering: Group same-day news by tag similarity
3. Event tracking: Cross-day tracking via tag overlap
"""
import uuid
from collections import defaultdict
from datetime import datetime, timezone, timedelta

from loguru import logger

from src.domain.market.intel.analysis.repository_interface import IAnalysisRepository
from src.domain.market.intel.repository_interface import IIntelRepository


class EventDetectionService:
    """Detects hot events from CCTV news burst patterns."""

    def __init__(
        self,
        analysis_repo: IAnalysisRepository,
        intel_repo: IIntelRepository,
    ):
        self.analysis_repo = analysis_repo
        self.intel_repo = intel_repo

    def detect_events(self) -> list[dict]:
        """Run event detection and return newly detected events.

        Returns:
            List of event dicts that were created or updated.
        """
        # Get recent CCTV news (last 24 hours)
        news_list = self.intel_repo.find_news_by_category(
            category="cctv_news", hours=24, limit=200,
        )
        if not news_list:
            return []

        # Group news by tags
        tag_groups = self._cluster_by_tags(news_list)

        # Detect burst groups (count > mean + 2*std)
        if not tag_groups:
            return []

        counts = [len(v) for v in tag_groups.values()]
        if len(counts) < 2:
            # Not enough variety to compute std
            return []

        import statistics
        mean_count = statistics.mean(counts)
        std_count = statistics.stdev(counts) if len(counts) > 1 else 0
        threshold = mean_count + 2 * std_count

        events = []
        for tag_key, group_news in tag_groups.items():
            if len(group_news) >= max(threshold, 3):
                event = self._create_or_update_event(tag_key, group_news)
                if event:
                    events.append(event)

        logger.info(
            f"[EventDetection] Detected {len(events)} hot events "
            f"from {len(news_list)} news items")
        return events

    def _cluster_by_tags(
        self, news_list: list[dict],
    ) -> dict[str, list[dict]]:
        """Group news by shared tags into clusters."""
        tag_groups: dict[str, list[dict]] = defaultdict(list)

        for news in news_list:
            tags = news.get("tags", [])
            if not tags:
                continue
            # Use the most specific tag (longest) as cluster key
            primary_tag = max(tags, key=len) if tags else None
            if primary_tag:
                tag_groups[primary_tag].append(news)

        return dict(tag_groups)

    def _create_or_update_event(
        self, tag_key: str, group_news: list[dict],
    ) -> dict | None:
        """Create a new event or update an existing one."""
        try:
            news_ids = [str(n["id"]) for n in group_news]
            titles = [n["title"] for n in group_news[:3]]
            title = " / ".join(titles[:2])
            if len(group_news) > 2:
                title += f" 等{len(group_news)}篇"

            all_tags = []
            for n in group_news:
                all_tags.extend(n.get("tags", []))
            keywords = list(set(all_tags))[:5]

            published_times = [
                n["published_at"] for n in group_news
                if n.get("published_at")
            ]
            start_time = min(published_times) if published_times else None
            last_seen = max(published_times) if published_times else None

            # Compute heat score: normalize by group size
            heat_score = min(len(group_news) / 20.0, 1.0)

            # Determine trend
            trend = "rising"
            if len(group_news) >= 5:
                trend = "rising"
            elif len(group_news) >= 3:
                trend = "stable"

            event_id = self.analysis_repo.upsert_event({
                "title": title,
                "keywords": keywords,
                "news_ids": news_ids,
                "start_time": (
                    datetime.fromisoformat(start_time)
                    if isinstance(start_time, str) else start_time
                ) if start_time else datetime.now(timezone.utc),
                "last_seen": (
                    datetime.fromisoformat(last_seen)
                    if isinstance(last_seen, str) else last_seen
                ) if last_seen else datetime.now(timezone.utc),
                "heat_score": heat_score,
                "trend": trend,
            })

            return {
                "id": event_id,
                "title": title,
                "heat_score": heat_score,
                "news_count": len(group_news),
            }
        except Exception as e:
            logger.warning(f"[EventDetection] Failed to create event: {e}")
            return None

    def get_active_events(self, limit: int = 20) -> list[dict]:
        """Get currently active hot events."""
        return self.analysis_repo.find_active_events(limit=limit)

    def get_event_detail(self, event_id: str) -> dict | None:
        """Get event detail with associated news."""
        return self.analysis_repo.find_event_by_id(event_id)

"""News-market price correlation analysis.

Analyzes the relationship between CCTV news events and
sector/index price movements within a time window.
"""
from datetime import datetime, timezone, timedelta

from loguru import logger

from src.domain.market.intel.analysis.repository_interface import IAnalysisRepository
from src.domain.market.intel.repository_interface import IIntelRepository


class MarketCorrelationService:
    """Correlates CCTV news with market price movements."""

    def __init__(
        self,
        analysis_repo: IAnalysisRepository,
        intel_repo: IIntelRepository,
    ):
        self.analysis_repo = analysis_repo
        self.intel_repo = intel_repo

    def compute_correlations(self, hours: int = 4) -> int:
        """Compute news-market correlations for recent CCTV news.

        Args:
            hours: Time window to look back for news.

        Returns:
            Number of correlations computed.
        """
        news_list = self.intel_repo.find_news_by_category(
            category="cctv_news", hours=hours, limit=50,
        )
        if not news_list:
            return 0

        correlations = []
        for news in news_list:
            metadata = news.get("metadata") or {}
            sectors = metadata.get("related_sectors", [])
            impact = metadata.get("market_impact", "None")

            if impact == "None" or not sectors:
                continue

            # Compute a simple correlation score based on
            # sentiment and market_impact level
            sentiment = news.get("sentiment", 0.0) or 0.0
            impact_scores = {"High": 0.8, "Medium": 0.5, "Low": 0.3}
            base_score = impact_scores.get(impact, 0.2)

            correlation_score = sentiment * base_score

            sector_impact = {}
            for sector in sectors[:5]:
                # Simplified: no real price data lookup here
                # Real implementation would query stock_ohlcv table
                sector_impact[sector] = round(correlation_score, 3)

            correlations.append({
                "news_id": news["id"],
                "related_stocks": [],
                "sector_impact": sector_impact,
                "correlation_score": round(correlation_score, 3),
            })

        if correlations:
            saved = self.analysis_repo.save_market_correlations(correlations)
            logger.info(
                f"[MarketCorrelation] Computed {saved} correlations "
                f"from {len(news_list)} news items")
            return saved
        return 0

    def get_market_impacts(self, hours: int = 24, limit: int = 20) -> list[dict]:
        """Get recent news-market impact records."""
        return self.analysis_repo.find_market_impacts(
            hours=hours, limit=limit)

    def get_news_impact(self, news_id: int) -> dict | None:
        """Get market impact for a specific news item."""
        return self.analysis_repo.find_market_impact_by_news_id(news_id)
